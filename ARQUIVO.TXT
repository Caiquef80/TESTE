#ALTERAÇÕES FEITAS PELA WEB
#SEGUNDA MUDANÇA FEITA PELA WEB
#INICIALIZAR UM REPOSITORIO 
GIT INIT .

#Mostrar o estado atual do repositorio
git status

#ADICIONAR UM ARQUIVO EM STAGE
git add <nome_do_arquivo>

#REMOVER ARQUIVO DO STAGE
git rm --cached <nome_do_arquivo>

#RESTAURA UM ARQUIVO
git restore <nome_do_arquivo>

#SALVAR O ESTADO DO REPOSITORIO
git commit -m "sua msg"

#MOSTRAR O HISTORICO DE COMMITS
git log
git log --oneline

#MOSTRAR AS ALTERAÇÕES NO REPOSITORIO
git diff 

#EXIBIR AS ALTERAÇÕES DE UM COMMIT
git show <hash>

#LISTAR AS BRANCHS

#RENOMEAR A BRANCH ATUAL
git branch -m <nome_da_branch>

#CRIAR UMA NOVA BRANCH

git branch <nome_da_branch>

#DELETAR UMA BRANCH
git branch -d <nome_da_branch>

#TROCAR UMA BRANCH
git switch <nome_da_branch>

#MESCLAR UMA BRANCH NA OUTRA
git merge <nome_da_branch>

#MOSTRAR AUTORES DE UM ARQUIVO
git blame <nome_do_arquivo>

#ADICIONAR TODOS OS ARQUIVOS NO STAGE

git add .

#DELETAR FORÇADAMENTE UMA BRANCH
git branch -D <nome_da_branch>

#COMMITAR E SUBSTITUIR O ULTIMO ARQUIVO
git commit --amend

#VOLTAR PARA UM COMMIT ANTERIOR(LOCAL)
git reset <hash_do_commit>

#DESFAZER UM COMMIT (REMOTO)
git revert <hash_do_commit>

#CLONAR UM REPOSITORIO REMOTO
git clone <URL>

#VERIFICAR ENDEREÇO REMOTO 
git remote -v

#ADICIONAR ENDEREÇO REMOTO EM UM REPOSITORIO LOCAL
git remote add <url>

#ENVIAR ALTERAÇOES E DEFINIR A UPSTREAM(REMOTO) COM A BRANCH LOCAL
git push -u origin <nome_da_branch>

#ENVIAR ALTERAÇOES PARA O GIT HUB
git push
git push origin <nome_da_branch>

#ENVIAR ALTERAÇOES DE UMA BRANCH LOCAL PARA UMA BRANCH REMOTA
git push origin <nome_da_branch_local>:<nome_da_branch_remoto>


#BAIXAR AS ATUALIZAÇÕES DO REPOSITORIO REMOTO SEM MESCLAR
git fetch <nome_da_branch>

#BAIXAR AS ATUALIZAÇÕES DO REPOSITORIO REMOTO E MESCLAR
git pull <nome_da_branch>
